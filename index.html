<body>


    <h1>🪟 Window Cleaning Planner</h1>

    

    <!-- Step 1: Permission Box -->

    <div class="card">

        <h2>1. Enable Alerts</h2>

        <button id="auth-btn">Allow Notifications</button>

    </div>


    <!-- Step 2: Add Job Form -->

    <div class="card">

        <h2>2. Add New Job</h2>

        <label for="client">Client Name / Address:</label>

        <input type="text" id="client" placeholder="e.g., Smith - 123 Main St">


        <label for="job-date">Job Date & Time:</label>

        <input type="datetime-local" id="job-date">


        <button id="save-btn">Add to Schedule</button>

    </div>


    <!-- Step 3: Job List -->

    <div class="card">

        <h2>📅 Upcoming Schedule</h2>

        <div id="job-list">No jobs scheduled yet.</div>

    </div>


    <script>

        const authBtn = document.getElementById('auth-btn');

        const saveBtn = document.getElementById('save-btn');

        const clientInput = document.getElementById('client');

        const dateInput = document.getElementById('job-date');

        const jobListDiv = document.getElementById('job-list');


        let jobs = JSON.parse(localStorage.getItem('cleaning_jobs')) || [];


        // Request Permission

        authBtn.addEventListener('click', () => {

            Notification.requestPermission().then(permission => {

                if (permission === 'granted') {

                    alert('Notifications enabled successfully!');

                    authBtn.textContent = '✅ Notifications Enabled';

                    authBtn.disabled = true;

                }

            });

        });


        // Update Button text if already granted

        if (Notification.permission === 'granted') {

            authBtn.textContent = '✅ Notifications Enabled';

            authBtn.disabled = true;

        }


        // Save a job

        saveBtn.addEventListener('click', () => {

            const client = clientInput.value.trim();

            const dateStr = dateInput.value;


            if (!client || !dateStr) {

                alert('Please fill out both fields!');

                return;

            }


            const job = {

                id: Date.now(),

                client: client,

                dateTime: dateStr,

                notified: false

            };


            jobs.push(job);

            localStorage.setItem('cleaning_jobs', JSON.stringify(jobs));

            

            clientInput.value = '';

            dateInput.value = '';

            

            renderJobs();

        });


        // Delete a job

        window.deleteJob = (id) => {

            jobs = jobs.filter(job => job.id !== id);

            localStorage.setItem('cleaning_jobs', JSON.stringify(jobs));

            renderJobs();

        };


        // Render job list to screen

        function renderJobs() {

            if (jobs.length === 0) {

                jobListDiv.innerHTML = 'No jobs scheduled yet.';

                return;

            }


            // Sort jobs by closest date first

            jobs.sort((a, b) => new Date(a.dateTime) - new Date(b.dateTime));


            jobListDiv.innerHTML = jobs.map(job => {

                const formattedDate = new Date(job.dateTime).toLocaleString([], {

                    dateStyle: 'short', 

                    timeStyle: 'short'

                });

                return `

                    <div class="job-item">

                        <div>

                            <strong>${job.client}</strong><br>

                            <small>📅 ${formattedDate}</small>

                        </div>

                        <button class="delete-btn" onclick="deleteJob(${job.id})">Remove</button>

                    </div>

                `;

            }).join('');

        }


        // ⏰ Checks every 10 seconds for 1-week alerts

        function checkReminders() {

            if (Notification.permission !== 'granted') return;


            const now = new Date();

            const ONE_WEEK_MS = 604800000; // 7 days in milliseconds


            let updated = false;


            jobs.forEach(job => {

                const jobTime = new Date(job.dateTime);

                const timeDiff = jobTime - now;


                // Trigger alert if the job is less than 7 days away and hasn't triggered yet

                if (timeDiff > 0 && timeDiff <= ONE_WEEK_MS && !job.notified) {

                    new Notification("🪟 Window Cleaning Reminder!", {

                        body: `1 week until job for: ${job.client}`,

                        requireInteraction: true 

                    });

                    job.notified = true; 

                    updated = true;

                }

            });


            if (updated) {

                localStorage.setItem('cleaning_jobs', JSON.stringify(jobs));

            }

        }


        // Initial run

        renderJobs();

        setInterval(checkReminders, 10000); 

    </script>
